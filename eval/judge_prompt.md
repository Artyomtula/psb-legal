{"key": "<key из пакета>", "correctness": <0-5>, "grounding": <0-5>, "reliability": <0-5>, "completeness": <0-5>, "clarity": <0-5>, "overall": <0-10>, "critical": ["..."], "comment": "одно-два предложения"}
